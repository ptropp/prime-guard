// ============================== // PRIME GUARD - WEBSITE SCRIPT // ============================== // Your Discord application client ID const CLIENT_ID = "154008590462025738"; // Your GitHub Pages website const REDIRECT_URI = "https://ptrop.github.io/prime-guard/"; // Discord OAuth scopes const SCOPES = "identify guilds"; // Build the Discord login URL const discordLoginURL = `https://discord.com/oauth2/authorize` + `?client_id=${CLIENT_ID}` + `&response_type=code` + `&redirect_uri=${encodeURIComponent(REDIRECT_URI)}` + `&scope=${encodeURIComponent(SCOPES)}`; // ============================== // ADD TO DISCORD BUTTONS // ============================== function connectDiscord() { window.location.href = discordLoginURL; } // ============================== // PLAN SELECTION // ============================== function choosePlan(plan) { localStorage.setItem("primeGuardPlan", plan); if (plan === "free") { alert("FREE plan selected! Now connect your Discord account."); } if (plan === "plus") { alert("PRIME PLUS selected! Payment setup will be added next."); } if (plan === "pro") { alert("PRIME PRO selected! Payment setup will be added next."); } } // ============================== // WAIT FOR PAGE TO LOAD // ============================== document.addEventListener("DOMContentLoaded", () => { // Add to Discord buttons const addToDiscordButtons = document.querySelectorAll( ".add-button, .primary-button" ); addToDiscordButtons.forEach(button => { button.addEventListener("click", (event) => { event.preventDefault(); connectDiscord(); }); }); // FREE plan button const freeButton = document.querySelector( "#free-button, .free-button" ); if (freeButton) { freeButton.addEventListener("click", (event) => { event.preventDefault(); choosePlan("free"); setTimeout(() => { connectDiscord(); }, 500); }); } // PLUS plan button const plusButton = document.querySelector( "#plus-button, .plus-button" ); if (plusButton) { plusButton.addEventListener("click", (event) => { event.preventDefault(); choosePlan("plus"); setTimeout(() => { connectDiscord(); }, 500); }); } // PRO plan button const proButton = document.querySelector( "#pro-button, .pro-button" ); if (proButton) { proButton.addEventListener("click", (event) => { event.preventDefault(); choosePlan("pro"); setTimeout(() => { connectDiscord(); }, 500); }); } });
